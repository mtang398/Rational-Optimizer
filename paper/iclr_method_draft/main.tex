\section{Rational Matrix Policy Optimization}
\label{sec:method}

Let $\mathcal{L}(\theta)$ be the language-model training objective.  For layer
$l$, the RLB block projects the residual-stream input $x_l\in\mathbb{R}^d$ into
$G$ groups of width $m$,
\begin{align}
    z_l=A_lx_l, \qquad z_{l,g}=\operatorname{group}_g(z_l),
\end{align}
normalizes each group by its RMS radius,
\begin{align}
    r_{l,g}=\sqrt{m^{-1}\lVert z_{l,g}\rVert_2^2+\epsilon_{\rm rms}},
    \qquad u_{l,g}=z_{l,g}/r_{l,g},
\end{align}
and applies a learned rational map in normalized coordinates,
\begin{align}
    h_{l,g}=r_{l,g}R_{l,g}(u_{l,g}),
    \qquad
    y_l=B_l\operatorname{concat}_{g=1}^G h_{l,g}.
\end{align}
The input matrix $A_l$ selects rational domains, the coefficients of $R_{l,g}$
shape the group response, and the output matrix $B_l$ recombines rational
features.  MatrixPolicy uses this separation by partitioning parameters as
\begin{align}
    \theta=\theta_{\mathrm{base}}\cup\theta_{\mathrm{coeff}}
    \cup\{A_l,B_l\}_{l=1}^L.
\end{align}
The base parameters use the same optimizer as the non-rational baseline.  Rational
coefficients use a coefficient rule.  MatrixPolicy is applied only to $A_l$ and
$B_l$.

In the homogeneous case $\epsilon_{\rm rms}=0$, and for nonzero group radii, each
group has a positive scale gauge:
\begin{align}
    A_{l,g}\mapsto a_gA_{l,g},\qquad
    B_{l,g}\mapsto a_g^{-1}B_{l,g},\qquad a_g>0.
\end{align}
The represented function is unchanged because $z_{l,g}$ and $r_{l,g}$ scale
together, $u_{l,g}$ is unchanged, and the inverse scale in $B_l$ cancels the scale
in $h_{l,g}$.  With a nonzero RMS floor this symmetry is only approximate, so we use the
gauge as an optimizer-design principle rather than as a formal invariance claim.  This
matters because optimizer state is stored in coordinates, not in the quotient space of
represented functions.

At step $t$, each matrix role $r\in\{\mathrm{in},\mathrm{out}\}$ receives a
role-depth factor $\rho_r(l)$ and an on-policy statistic factor
$s_{\mathrm{stat}}(l,r,t)$.  The matrix update is
\begin{align}
    M_{l,r}^{t+1}=M_{l,r}^{t}
    &+\Delta_{\mathrm{adapt}}\!\left(M_{l,r};
      \eta_t a_{\mathrm{mat}}(l,r,t)[1-\mu(l,r,t)]\right) \nonumber\\
    &+\Delta_{\mathrm{mat}}\!\left(M_{l,r};
      \eta_t a_{\mathrm{matnorm}}\mu(l,r,t)\right),
\end{align}
where $M_{l,\mathrm{in}}=A_l$, $M_{l,\mathrm{out}}=B_l$, $\eta_t$ is the shared
base learning rate, $\Delta_{\mathrm{adapt}}$ is an Adam-style adaptive step, and
$\Delta_{\mathrm{mat}}$ is a short early matrix-normalized step.  The mixture
coefficient $\mu(l,r,t)$ is active only early in training.  After the matrix
update, MatrixPolicy computes a target log-ratio for each group and applies the
bounded rebalance
\begin{align}
    A_{l,g}\leftarrow e^{\ell_g}A_{l,g},\qquad
    B_{l,g}\leftarrow e^{-\ell_g}B_{l,g}.
\end{align}
The update changes the representative used by the optimizer while preserving, or
with $\epsilon_{\rm rms}>0$ approximately preserving, the represented block
function.  Full hyperparameter definitions are in Appendix~\ref{app:details}.

\begin{center}
\fbox{%
\begin{minipage}{0.92\linewidth}
\textbf{Algorithm 1: Rational Matrix Policy step.}
Given gradients $\nabla\mathcal{L}(\theta^t)$ and base learning rate $\eta_t$:
\begin{enumerate}
    \item Update exponential moving averages of rational-group pressures.
    \item Compute role-depth and group-stat multipliers for each RLB matrix.
    \item Update non-rational parameters with the base optimizer and update
    rational coefficients with the coefficient rule.
    \item Update each $A_l$ and $B_l$ with the MatrixPolicy mixture step.
    \item Apply the bounded positive-gauge rebalance to each rational group.
\end{enumerate}
\end{minipage}}
\end{center}

\section{MatrixPolicy Details}
\label{app:details}

\paragraph{Role-depth factors.}
For layers indexed by $l=1,\ldots,L$ with $L>1$, define normalized layer depth
$d_l=(l-1)/(L-1)$ and set
\begin{align}
    \rho_{\mathrm{in}}(l)&=\clip(1-0.50(d_l-0.5),0.55,1.40),\\
    \rho_{\mathrm{out}}(l)&=\clip(1+1.00(d_l-0.5),0.55,1.40),\\
    a_{\mathrm{role}}(l,r)&=\max(0.10,1+1.20(\rho_r(l)-1)).
\end{align}
Let $\xi_t=t/T$ be training progress and let $\sigma_{a,b}(\xi)$ denote the
smoothstep gate from $0$ to $1$ over $[a,b]$.  The statistic multiplier
$a_{\mathrm{stat}}(l,r,t)>0$ is the layer/role aggregate of the group
preconditioners defined below, normalized to one when group statistics are not
active.  The local adaptive scale is
\begin{align}
    a_{\mathrm{mat}}(l,r,t)=\clip(3.0\,a_{\mathrm{role}}(l,r)a_{\mathrm{stat}}(l,r,t),0.40,4.0).
\end{align}
The matrix-normalized branch uses multiplier $a_{\mathrm{matnorm}}>0$ and
mixture weight $\mu(l,r,t)\in[0,1]$, a smoothstep-gated function of
$\xi_t$, layer depth, matrix role, and $a_{\mathrm{stat}}(l,r,t)$.

\paragraph{On-policy group statistics.}
All ratios and logarithms below are evaluated on strictly positive stabilized
RMS, EMA, and gain quantities; the displayed formulas omit the corresponding
epsilon floors.
For group $g$, MatrixPolicy tracks relative gradient pressures
\begin{align}
    p_{\mathrm{in},g}=\frac{\rms(\nabla A_{l,g})}{\rms(A_{l,g})},\qquad
    p_{\mathrm{out},g}=\frac{\rms(\nabla B_{l,g})}{\rms(B_{l,g})},\qquad
    p_{\mathrm{rat},g}=\rms(\nabla\theta_{\mathrm{coeff},g}).
\end{align}
With EMAs $q_{\mathrm{in},g}$, $q_{\mathrm{out},g}$, and $q_{\mathrm{rat},g}$,
we define
\begin{align}
    \pi_g=\log q_{\mathrm{in},g}-\log q_{\mathrm{out},g},\qquad
    \alpha_g=\log q_{\mathrm{rat},g}-\frac{1}{2}(\log q_{\mathrm{in},g}+\log q_{\mathrm{out},g}).
\end{align}

\paragraph{Group-stat preconditioning.}
The group preconditioner is centered and clipped:
\begin{align}
    c_g=c_{\mathrm{gain},g}c_{\mathrm{pressure},g}c_{\mathrm{activity},g},
    \qquad c_g\leftarrow c_g/\geom(c),
    \qquad c_g\leftarrow\clip(c_g,0.75,1.35).
\end{align}
Here $k_g>0$ is the role-specific group gain used for matrix preconditioning: the
derivative-RMS gain for $A_l$ and the output-RMS gain for $B_l$.  The factors are
\begin{align}
    c_{\mathrm{gain},g}=\left(\frac{\geom(k)}{k_g}\right)^{0.20},\qquad
    c_{\mathrm{pressure},g}=\exp(0.10\,d_{\mathrm{pressure},g}),
\end{align}
where $d_{\mathrm{pressure},g}=-\pi_g$ for $A_l$ and $+\pi_g$ for $B_l$. The
activity damping term is
\begin{align}
    c_{\mathrm{activity},g}=\exp[-0.20\,\relu((\alpha_g-0.05)/0.45)].
\end{align}

\paragraph{Gauge rebalance.}
Let
\begin{align}
    \gamma_g=\log\rms(A_{l,g})-\log\rms(B_{l,g}).
\end{align}
The target log-ratio is
\begin{align}
    \tau_g=\frac{1}{2}(\log d^{\mathrm{gain}}_g-\log o^{\mathrm{gain}}_g)
    +\beta_{\mathrm{pressure}}(\log q_{\mathrm{in},g}-\log q_{\mathrm{out},g}).
\end{align}
Here $d^{\mathrm{gain}}_g>0$ and $o^{\mathrm{gain}}_g>0$ are the derivative and
output group-gain estimates, and $\beta_{\mathrm{pressure}}$ is the pressure
balance coefficient.  Let $s(t,l)\in[0,1]$ be the rebalance schedule,
$a^{\mathrm{activity}}_g\in(0,1]$ the activity damping multiplier, and
$\ell_{\max}>0$ the maximum absolute log-scale move.  The exact target step is
$\ell_g^{\star}=\frac{1}{2}(\tau_g-\gamma_g)$, while the applied bounded move is
\begin{align}
    \ell_g=\clip(s(t,l)a^{\mathrm{activity}}_g\ell_g^{\star},-\ell_{\max},\ell_{\max}).
\end{align}
For true RMS values, the clipped update maps $\gamma_g$ to $\gamma_g+2\ell_g$ and
is a bounded move toward $\tau_g$ rather than exact canonicalization unless clipping and
schedule damping are inactive.  With stabilized RMS values this identity is approximate;
the implementation uses the bounded move as a controlled representative update for the RLB
matrices.
