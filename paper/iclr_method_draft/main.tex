\section{Method}
\label{sec:method}

Let $\mathcal{L}(\theta)$ be the language-model training objective. We study
optimizers for Transformer feed-forward layers whose hidden nonlinearity is a
grouped rational map. The method, Rational Matrix Policy Optimization, keeps a
standard base optimizer for non-rational parameters and replaces only the update
rule for the matrices surrounding each rational block. Its design uses two facts:
the input and output matrices of a rational block have different functional
roles, and the block has an exact positive gauge symmetry.

\subsection{Rational feed-forward parameterization}
\label{subsec:rlb-main}

For layer $l$, let $x_l\in\mathbb{R}^d$ be the residual-stream input. The RLB
block projects $x_l$ into $G$ groups of width $m$,
\begin{align}
    z_l=A_lx_l, \qquad z_{l,g}=\operatorname{group}_g(z_l),
\end{align}
normalizes each group by its RMS radius,
\begin{align}
    r_{l,g}=\sqrt{m^{-1}\lVert z_{l,g}\rVert_2^2+\epsilon},
    \qquad u_{l,g}=z_{l,g}/r_{l,g},
\end{align}
and applies a learned rational map in normalized coordinates,
\begin{align}
    h_{l,g}=r_{l,g}R_{l,g}(u_{l,g}),
    \qquad
    y_l=B_l\operatorname{concat}_{g=1}^G h_{l,g}.
\end{align}
The parameters around the rational map therefore have distinct meanings:
$A_l$ selects the domains seen by rational groups, $R_{l,g}$ controls nonlinear
shape, and $B_l$ recombines rational features into the residual stream.

\subsection{Gauge symmetry}
\label{subsec:gauge-main}

The group normalization makes the block invariant to a positive reparameterization.
For $a_g>0$, let $D_l(a)$ be block diagonal with $g$th block $a_gI_m$. Then
\begin{align}
    A_l' = D_l(a)A_l, \qquad B_l' = B_lD_l(a)^{-1}
\end{align}
represents the same function as $(A_l,B_l)$. Indeed, $z_{l,g}$ and $r_{l,g}$ both
scale by $a_g$, while $u_{l,g}$ is unchanged and the inverse scale in $B_l'$ cancels
the change in $h_{l,g}$. Consequently, parameter motion and function motion are
not equivalent: an optimizer can change matrix norms and adaptive states while
leaving the represented function fixed.

\subsection{MatrixPolicy update}
\label{subsec:matrixpolicy-main}

We partition parameters as
\begin{align}
    \theta=\theta_{\mathrm{base}}\cup\theta_{\mathrm{coeff}}
    \cup\{A_l,B_l\}_{l=1}^L.
\end{align}
The base parameters are updated by a standard optimizer, and rational coefficients
are updated by a coefficient rule. MatrixPolicy is applied only to $A_l$ and
$B_l$. At step $t$, each matrix role $r\in\{\mathrm{in},\mathrm{out}\}$ receives a
role-depth factor $\rho_r(l)$ and an on-policy statistic factor $s_{\mathrm{stat}}(l,t)$.
The matrix update is
\begin{align}
    M_{l,r}^{t+1}=M_{l,r}^{t}
    +\Delta_{\mathrm{adapt}}\!\left(M_{l,r};\eta_t a_{\mathrm{mat}}(l,r,t)[1-\mu(l,r,t)]\right)
    +\Delta_{\mathrm{mat}}\!\left(M_{l,r};\eta_t a_{\mathrm{matnorm}}\mu(l,r,t)\right),
\end{align}
where $M_{l,\mathrm{in}}=A_l$, $M_{l,\mathrm{out}}=B_l$, $\eta_t$ is the shared
base learning rate, $\Delta_{\mathrm{adapt}}$ is an Adam-style adaptive step
\citep{kingma2015iclr-adam}, and $\Delta_{\mathrm{mat}}$ is a short early
matrix-normalized step. The mixture coefficient $\mu(l,r,t)$ is nonzero only in
the early phase of training. After the matrix update, the optimizer applies the
exact gauge transform
\begin{align}
    A_{l,g}\leftarrow e^{\ell_g}A_{l,g},\qquad
    B_{l,g}\leftarrow B_{l,g}/e^{\ell_g},
\end{align}
which changes the parameterization but not the represented function. The detailed
forms of $\rho_r$, $\mu$, group-stat scaling, and $\ell_g$ are given in
Appendix~\ref{app:details}.

\subsection{Algorithm}
\label{subsec:algorithm-main}

\begin{center}
\fbox{%
\begin{minipage}{0.92\linewidth}
\textbf{Algorithm 1: Rational Matrix Policy step.}
Given gradients $\nabla\mathcal{L}(\theta^t)$ and base learning rate $\eta_t$:
\begin{enumerate}
    \item Update exponential moving averages of rational-group pressures.
    \item Compute role-depth and group-stat multipliers for each RLB matrix.
    \item Update non-rational parameters with the base optimizer and update
    rational coefficients with the coefficient rule.
    \item Update each $A_l$ and $B_l$ with the MatrixPolicy mixture step.
    \item Apply the exact positive-gauge rebalance to each rational group.
\end{enumerate}
\end{minipage}}
\end{center}

\section{MatrixPolicy Details}
\label{app:details}

\paragraph{Role-depth factors.}
For normalized layer depth $d_l=l/(L-1)$,
\begin{align}
    \rho_{\mathrm{in}}(l)&=\clip(1-0.50(d_l-0.5),0.55,1.40),\\
    \rho_{\mathrm{out}}(l)&=\clip(1+1.00(d_l-0.5),0.55,1.40),\\
    a_{\mathrm{role}}(l,r)&=\max(0.10,1+1.20(\rho_r(l)-1)).
\end{align}
The local adaptive scale is
\begin{align}
    a_{\mathrm{mat}}(l,r,t)=\clip(3.0\,a_{\mathrm{role}}(l,r)a_{\mathrm{stat}}(l,r,t),0.40,4.0).
\end{align}

\paragraph{On-policy group statistics.}
For group $g$, MatrixPolicy tracks relative gradient pressures
\begin{align}
    p_{\mathrm{in},g}=\frac{\rms(\nabla A_{l,g})}{\rms(A_{l,g})},\qquad
    p_{\mathrm{out},g}=\frac{\rms(\nabla B_{l,g})}{\rms(B_{l,g})},\qquad
    p_{\mathrm{rat},g}=\rms(\nabla\theta_{\mathrm{coeff},g}).
\end{align}
With EMAs $q_{\mathrm{in},g}$, $q_{\mathrm{out},g}$, and $q_{\mathrm{rat},g}$,
we define
\begin{align}
    \pi_g=\log q_{\mathrm{in},g}-\log q_{\mathrm{out},g},\qquad
    \alpha_g=\log q_{\mathrm{rat},g}-\frac{1}{2}(\log q_{\mathrm{in},g}+\log q_{\mathrm{out},g}).
\end{align}

\paragraph{Group-stat preconditioning.}
The group preconditioner is centered and clipped:
\begin{align}
    c_g=c_{\mathrm{gain},g}c_{\mathrm{pressure},g}c_{\mathrm{activity},g},
    \qquad c_g\leftarrow c_g/\geom(c),
    \qquad c_g\leftarrow\clip(c_g,0.75,1.35).
\end{align}
The factors are
\begin{align}
    c_{\mathrm{gain},g}=\left(\frac{\geom(k)}{k_g}\right)^{0.20},\qquad
    c_{\mathrm{pressure},g}=\exp(0.10\,d_{\mathrm{pressure},g}),
\end{align}
where $d_{\mathrm{pressure},g}=-\pi_g$ for $A_l$ and $+\pi_g$ for $B_l$. The
activity damping term is
\begin{align}
    c_{\mathrm{activity},g}=\exp[-0.20\,\relu((\alpha_g-0.05)/0.45)].
\end{align}

\paragraph{Gauge rebalance.}
Let
\begin{align}
    \gamma_g=\log\rms(A_{l,g})-\log\rms(B_{l,g}).
\end{align}
The target log-ratio is
\begin{align}
    \tau_g=\frac{1}{2}(\log d^{\mathrm{gain}}_g-\log o^{\mathrm{gain}}_g)
    +\beta_{\mathrm{pressure}}(\log q_{\mathrm{in},g}-\log q_{\mathrm{out},g}).
\end{align}
The applied correction is
\begin{align}
    \ell_g=\frac{1}{2}(\tau_g-\gamma_g),\qquad
    \ell_g\leftarrow\clip(s(t,l)a^{\mathrm{activity}}_g\ell_g,-\ell_{\max},\ell_{\max}).
\end{align}
The transform $A_{l,g}\leftarrow e^{\ell_g}A_{l,g}$ and
$B_{l,g}\leftarrow B_{l,g}/e^{\ell_g}$ preserves the RLB function exactly up to
floating-point error.
