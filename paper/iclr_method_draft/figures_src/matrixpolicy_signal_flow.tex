% Step 1 design document.
% Goal: show that MatrixPolicy is a selective optimizer route, not a general
% optimizer replacement.  The top rail is the only special path and is reserved
% for the RLB matrix pair A_l,B_l.  The bottom rail is a muted AdamW-only bypass
% for rational coefficients and ordinary Transformer tensors.  Exact update
% equations remain in the method text; the figure emphasizes routing, gates,
% transient Muon, and reciprocal pair balancing without cramped formulas.

    minimum height=0.74cm, align=center, inner sep=4pt},
  station/.style={rectangle, rounded corners=4pt, fill=white, line width=0.82pt,
    minimum height=0.92cm, align=center, inner sep=4pt},

    fill=white, line width=0.88pt, minimum width=1.78cm, minimum height=0.58cm,

  adamOnly/.style={rectangle, rounded corners=4pt, draw=mpMuted,
    fill=mpGreyFill, line width=0.72pt, minimum width=1.50cm,

  read/.style={-{Stealth[length=3pt 1.0, width'=0pt 0.48]},
    densely dashed, line width=0.58pt, shorten >=1pt, shorten <=1pt,

}
