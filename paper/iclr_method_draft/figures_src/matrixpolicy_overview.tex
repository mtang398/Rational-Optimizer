% Step 1 design document.
% Domain / format / density: ML method overview, TikZ standalone, medium density.
% Five-second message: MatrixPolicy observes detached summaries and writes only
% Delta A_l / Delta B_l, never the activation stream itself.
% Layout: left-to-right forward path A_l -> normalized grouped rational response
% -> B_l, with the detached summary bus below a clean stop-gradient boundary.
% MatrixPolicy sits below the bus; one purple write bus exits its Delta-only box
% and terminates at visible write ports on the two matrix handles.
% Modules: A_l matrix glyph, RLB response card, B_l matrix glyph, four detached
% summary chips, MatrixPolicy, and a Delta-only update box.
% Space plan: keep the membrane label centered in a white break, keep summaries
% inside their panel, and route purple write rails below the policy boxes so no
% text is crossed by a line.

\newcommand{\matrixglyph}[3]{%
  \begin{scope}[shift={#1}]
    \draw[matrixFrame] (0,0) rectangle (1.72,1.18);
    \foreach \i in {0,...,3} {
      \foreach \j in {0,...,2} {
        \pgfmathsetmacro{\shade}{#3+6*\i+8*\j}
        \fill[mpBlue!\shade!white, draw=white, line width=0.35pt]
          ({0.22+0.32*\i},{0.22+0.25*\j}) rectangle ++(0.24,0.18);
      }
    }
    \draw[mpPurple, line width=1.02pt, rounded corners=1pt]
      #2;
  \end{scope}
}

\newcommand{\summarychip}[5]{%
  \node[chip, draw=#5] (#1) at (#2,#3) {};
  \node[font=\scriptsize\bfseries, text=#5] at (#2,{#3+0.14}) {#4};
}
