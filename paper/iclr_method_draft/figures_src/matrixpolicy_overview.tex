% Step 1 design document.
% Domain / format / density: ICLR method overview, TikZ standalone, medium density.
% Five-second message: RLB exposes a normalized rational matrix pair; MatrixPolicy
% reads detached summaries and writes only Delta A_l / Delta B_l.
% Layout: top forward lane A_l -> grouped rational response -> B_l; middle
% detached-summary shelf with four clearly separated cards; bottom MatrixPolicy
% controller with two direct write arrows to the two matrix handles.
% Visual hazards to avoid: dashed read arrows must not cross summary text, the
% stop-gradient line must remain visible, and write arrows must terminate at the
% A_l/B_l boxes instead of forming a decorative rail.

\newcommand{\matrixglyph}[3]{%
  \begin{scope}[shift={#1}]
    \draw[matrixFrame] (0,0) rectangle (1.72,1.18);
    \foreach \i in {0,...,3} {
      \foreach \j in {0,...,2} {
        \pgfmathsetmacro{\shade}{#3+6*\i+8*\j}
        \fill[mpBlue!\shade!white, draw=white, line width=0.35pt]
          ({0.22+0.32*\i},{0.22+0.25*\j}) rectangle ++(0.24,0.18);
      }
    }
    \draw[mpPurple, line width=1.02pt, rounded corners=1pt]
      #2;
  \end{scope}
}

\newcommand{\summarychip}[5]{%
  \node[chip, draw=#5] (#1) at (#2,#3) {};
  \node[font=\scriptsize\bfseries, text=#5] at (#2,{#3+0.14}) {#4};
}
