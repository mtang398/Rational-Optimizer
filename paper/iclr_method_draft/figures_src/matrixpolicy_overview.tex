% Step 1 design document.
% Goal: make the activation--optimizer interface visible in five seconds.
% The figure shows only three ideas: (i) A_l and B_l are the matrix handles
% around an RLB rational response, (ii) RLB exposes a stop-gradient
% summary bus, and (iii) MatrixPolicy reads that bus and writes only the two
% matrix handles.  The caption carries prose; the figure uses short labels,
% matrix glyphs, a visible detachment membrane, and direct write rails.

\newcommand{\matrixglyph}[3]{%
  \begin{scope}[shift={#1}]
    \draw[matrixFrame] (0,0) rectangle (1.72,1.18);
    \foreach \i in {0,...,3} {
      \foreach \j in {0,...,2} {
        \pgfmathsetmacro{\shade}{#3+6*\i+8*\j}
        \fill[mpBlue!\shade!white, draw=white, line width=0.35pt]
          ({0.22+0.32*\i},{0.22+0.25*\j}) rectangle ++(0.24,0.18);
      }
    }
    \draw[mpPurple, line width=1.02pt, rounded corners=1pt]
      #2;
  \end{scope}
}

\newcommand{\summarychip}[5]{%
  \node[chip, draw=#5] (#1) at (#2,#3) {};
  \node[font=\scriptsize\bfseries, text=#5] at (#2,{#3+0.14}) {#4};
}
