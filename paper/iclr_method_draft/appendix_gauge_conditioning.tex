\label{app:complete-opt-proof}
\label{app:gauge-conditioning}

\subsection{Quotient Geometry}
\label{app:complete-opt-proof:geometry}

For one RLB layer, suppress the layer index and write
\begin{align}
    z_g &= A_g x, \\
    r_g &= \sqrt{m^{-1}\norm{z_g}_2^2+\epsilon_{\rm rms}}, \\
    u_g &= z_g/r_g, \\
    h_g &= r_g R_g(u_g), \\
    y &= B\operatorname{concat}_{g=1}^G h_g .
\end{align}
For $a_g>0$, define
\begin{equation}
    T_a(A_g,B_g)=(a_gA_g,a_g^{-1}B_g),
    \qquad g=1,\ldots,G .
    \label{eq:complete-positive-action}
\end{equation}
Equivalently, with $s_g=\log a_g$,
\begin{equation}
    T_s(A_g,B_g)=(e^{s_g}A_g,e^{-s_g}B_g).
    \label{eq:complete-log-action}
\end{equation}
Let $\cP$ be the local RLB-matrix parameter space, let $\cG=(\R,+)^G$, and let
$\cQ=\cP/\cG$.

\begin{lemma}[Exact positive gauge]
\label{lem:exact-positive-gauge}
If $\epsilon_{\rm rms}=0$ and each zero-radius group, if present, is evaluated by a
bounded convention with $h_g=0$, then $T_s(A,B)$ and $(A,B)$ represent the same layer map
for all $s\in\R^G$.  In particular, the claim holds on any input for which all group
radii are nonzero.
\end{lemma}

\begin{proof}
For a group with $r_g>0$, $A_g'=e^{s_g}A_g$ gives $z_g'=e^{s_g}z_g$.  Since
$\epsilon_{\rm rms}=0$, $r_g'=e^{s_g}r_g$, hence $u_g'=u_g$ and
$h_g'=e^{s_g}h_g$.  The corresponding output block satisfies
$B_g'h_g'=e^{-s_g}B_g e^{s_g}h_g=B_gh_g$.  If $r_g=0$, then $z_g=0$ and also
$z_g'=0$; by the zero-radius convention both group features are zero, so the output
block is again unchanged.  Summing over groups gives the claim.
\end{proof}

\begin{proposition}[Stabilized gauge error]
\label{prop:stabilized-gauge-error}
Fix $|s_g|\leq S$ and $\epsilon_0>0$, and let $f_\epsilon$ denote the layer map with
radius $r_g=\sqrt{m^{-1}\norm{z_g}_2^2+\epsilon}$.  Suppose that, on the considered
local activation set,
\begin{equation}
    0<\sigma^2\leq m^{-1}\norm{z_g}_2^2\leq M^2<\infty
\end{equation}
for every group, the rational maps are bounded and Lipschitz on the resulting normalized
inputs uniformly for $0\leq\epsilon\leq\epsilon_0$ and $|s_g|\leq S$, and the output
blocks $B_g$ have bounded operator norm.  Then there is a constant $C$ independent of
$\epsilon$ such that, for all $0\leq\epsilon\leq\epsilon_0$,
\begin{equation}
    \norm{f_\epsilon(T_s(A,B);x)-f_\epsilon(A,B;x)}_2\leq C\epsilon .
    \label{eq:stabilized-gauge-error}
\end{equation}
\end{proposition}

\begin{proof}
It suffices to bound one group.  Let $a=e^{s_g}$ and
$S_z=m^{-1}\norm{z_g}_2^2$.  After applying $T_s$, the contribution of this group after
the inverse scaling of $B_g$ is governed by the radius
$\sqrt{S_z+\epsilon/a^2}$, whereas the original radius is $\sqrt{S_z+\epsilon}$.
Because $a$ is bounded above and below and $S_z\geq\sigma^2$,
\begin{equation}
    \left|\sqrt{S_z+\epsilon/a^2}-\sqrt{S_z+\epsilon}\right|\leq C_1\epsilon .
\end{equation}
The normalized inputs obey the corresponding bound
\begin{equation}
    \norm{z_g\left[(S_z+\epsilon/a^2)^{-1/2}-(S_z+\epsilon)^{-1/2}\right]}_2
    \leq C_2\epsilon,
\end{equation}
because $S_z\in[\sigma^2,M^2]$ and $a$ is bounded above and below.  Boundedness and
Lipschitzness of $R_g$, together with the bounded operator norm of $B_g$, therefore give
an $O(\epsilon)$ bound for the group output.
Summing over finitely many groups proves \eqref{eq:stabilized-gauge-error}.
\end{proof}

\begin{assumption}[Local quotient chart]
\label{ass:quotient-chart}
In a neighborhood of the trajectory, parameters can be written as
\begin{equation}
    \theta=\theta(q,s),
    \qquad q\in\R^d,
    \qquad s\in\R^G,
\end{equation}
where $q$ is a local coordinate on $\cQ$ and $s$ is the log-gauge coordinate.  The objective
satisfies
\begin{equation}
    L(\theta(q,s))=F(q).
    \label{eq:loss-quotient}
\end{equation}
For exact gauge identities, $\epsilon_{\rm rms}=0$.  For nonzero RMS stabilization,
Proposition~\ref{prop:stabilized-gauge-error} gives a local $O(\epsilon_{\rm rms})$
gauge error under nondegenerate activations; the remaining implementation effects are
included in the residual model of Assumption~\ref{ass:uniform-perturbation}.
\end{assumption}

\begin{lemma}[Gauge tangent orthogonality]
\label{lem:gauge-null-direction}
Under Lemma~\ref{lem:exact-positive-gauge}, for any differentiable loss depending on the
RLB matrices only through the represented function, with Frobenius inner products,
\begin{equation}
    \ip{\nabla_{A_g}L}{A_g}-\ip{\nabla_{B_g}L}{B_g}=0 .
    \label{eq:gauge-null-gradient}
\end{equation}
\end{lemma}

\begin{proof}
Differentiate $L(e^{s_g}A_g,e^{-s_g}B_g)=L(A_g,B_g)$ at $s_g=0$.
\end{proof}

\subsection{Canonicalized Quotient Preconditioning}
\label{app:complete-opt-proof:preconditioning}

For group $g$, define
\begin{equation}
    \gamma_g=\log \rms(A_g)-\log \rms(B_g).
    \label{eq:complete-gamma}
\end{equation}
The exact identities in this subsection use true positive RMS quantities.  For the
quadratic RMS floor $\widehat{\rms}_\epsilon(M)=\sqrt{\rms(M)^2+\epsilon}$, the active-floor
log-ratio error is locally $O(\epsilon)$, as shown in
Proposition~\ref{prop:stabilized-rms-ratio-error} below.  Given a target log-ratio
$\tau_g$, the unclipped rebalance is
\begin{equation}
    \ell_g=\frac{1}{2}(\tau_g-\gamma_g),
    \qquad
    A_g^+=e^{\ell_g}A_g,
    \qquad
    B_g^+=e^{-\ell_g}B_g.
    \label{eq:complete-rebalance}
\end{equation}

\begin{lemma}[RMS-ratio canonicalization]
\label{lem:complete-targeting}
For true positive RMS values, the update in \eqref{eq:complete-rebalance} satisfies
\begin{equation}
    \log\rms(A_g^+)-\log\rms(B_g^+)=\tau_g .
\end{equation}
\end{lemma}

\begin{proof}
By homogeneity of RMS under positive scaling,
\begin{align}
    \log\rms(A_g^+)-\log\rms(B_g^+)
    &=\log\rms(e^{\ell_g}A_g)-\log\rms(e^{-\ell_g}B_g) \\
    &=\gamma_g+2\ell_g
      =\tau_g .
\end{align}
\end{proof}

\begin{lemma}[Bounded rebalance moves toward the target]
\label{lem:bounded-rebalance-target}
Let $\gamma_g=\log\rms(A_g)-\log\rms(B_g)$ and let $\tau_g$ be the target log-ratio,
with true positive RMS values.  Suppose the applied log-scale move $\ell_g$ satisfies
\begin{equation}
    \ell_g(\tau_g-\gamma_g)\geq0,
    \qquad
    2|\ell_g|\leq |\tau_g-\gamma_g| .
    \label{eq:bounded-rebalance-condition}
\end{equation}
Then the rebalance $A_g^+=e^{\ell_g}A_g$, $B_g^+=e^{-\ell_g}B_g$ satisfies
\begin{equation}
    \left|\log\rms(A_g^+)-\log\rms(B_g^+)-\tau_g\right|
    \leq
    |\gamma_g-\tau_g| .
    \label{eq:bounded-rebalance-nonexpansive}
\end{equation}
If $\ell_g=\frac{1}{2}(\tau_g-\gamma_g)$, the new ratio is exactly $\tau_g$.
\end{lemma}

\begin{proof}
By RMS homogeneity, the new log-ratio is $\gamma_g^+=\gamma_g+2\ell_g$.  Let
$d=\tau_g-\gamma_g$.  If $d=0$, then \eqref{eq:bounded-rebalance-condition} implies
$\ell_g=0$ and the claim is immediate.  Otherwise, the same condition gives
$\ell_g=\operatorname{sign}(d)m$ for some $0\leq m\leq |d|/2$.  Hence
\begin{equation}
    |\gamma_g^+-\tau_g|=|2\ell_g-d|=|d|-2m\leq |d|=|\gamma_g-\tau_g|.
\end{equation}
The exact-target case follows by setting $m=|d|/2$.
\end{proof}

\begin{corollary}[Clipped scheduled rebalance]
\label{cor:clipped-scheduled-rebalance}
Let
\begin{equation}
    \ell_g^\star=\frac{1}{2}(\tau_g-\gamma_g),
    \qquad
    \ell_g=\clip(c_g\ell_g^\star,-\ell_{\max},\ell_{\max}),
\end{equation}
where $0\leq c_g\leq1$ and $\ell_{\max}>0$.  For true positive RMS values, the resulting
rebalance is nonexpansive toward the target in the sense of
\eqref{eq:bounded-rebalance-nonexpansive}.
\end{corollary}

\begin{proof}
The clipped value has the same sign as $\ell_g^\star$ and satisfies
$|\ell_g|\leq |\ell_g^\star|=|\tau_g-\gamma_g|/2$.  Lemma~\ref{lem:bounded-rebalance-target}
therefore applies.
\end{proof}

\begin{proposition}[Stabilized RMS-ratio error]
\label{prop:stabilized-rms-ratio-error}
Let $\widehat{\rms}_\epsilon(M)=\sqrt{\rms(M)^2+\epsilon}$.  Fix $|\ell_g|\leq S$ and
$\epsilon_0>0$.  If $\rms(A_g)\geq\sigma$ and $\rms(B_g)\geq\sigma$ for some
$\sigma>0$, then there is a constant $C$ independent of $\epsilon$ such that, for all
$0\leq\epsilon\leq\epsilon_0$,
\begin{align}
&\Bigl|\log\widehat{\rms}_\epsilon(e^{\ell_g}A_g)
      -\log\widehat{\rms}_\epsilon(e^{-\ell_g}B_g) \\
&\qquad\qquad
      -\bigl(\log\rms(A_g)-\log\rms(B_g)+2\ell_g\bigr)\Bigr|
\leq C\epsilon .
\label{eq:stabilized-rms-ratio-error}
\end{align}
\end{proposition}

\begin{proof}
For $R_A=\rms(A_g)$,
\begin{equation}
    \log\widehat{\rms}_\epsilon(e^{\ell_g}A_g)
    =\ell_g+\log R_A+\frac{1}{2}\log\left(1+\epsilon e^{-2\ell_g}R_A^{-2}\right).
\end{equation}
The last term is bounded by $C_A\epsilon$ uniformly because $|\ell_g|\leq S$ and
$R_A\geq\sigma$.  The same calculation for $B_g$ gives the second $O(\epsilon)$ term.
Combining the two bounds proves \eqref{eq:stabilized-rms-ratio-error}.
\end{proof}

\begin{lemma}[Gauge-dependent raw quotient step]
\label{lem:gauge-dependent-step}
Let $w>0$, $a=e^s\sqrt{w}$, $b=e^{-s}\sqrt{w}$, and
$\varphi(w)=\frac{1}{2}(w-y)^2$.  One Euclidean gradient-descent step in $(a,b)$ with step
size $\eta$ gives
\begin{equation}
    w^+=w-2\eta w\cosh(2s)(w-y)+\eta^2w(w-y)^2.
    \label{eq:scalar-gauge-step}
\end{equation}
\end{lemma}

\begin{proof}
Since $\partial\varphi/\partial a=(w-y)b$ and $\partial\varphi/\partial b=(w-y)a$,
\begin{align}
    w^+
    &=(a-\eta(w-y)b)(b-\eta(w-y)a) \\
    &=w-\eta(w-y)(a^2+b^2)+\eta^2(w-y)^2ab.
\end{align}
Using $ab=w$ and $a^2+b^2=w(e^{2s}+e^{-2s})=2w\cosh(2s)$ gives
\eqref{eq:scalar-gauge-step}.
\end{proof}

\begin{assumption}[Frozen quotient preconditioners for the quadratic specialization]
\label{ass:frozen-preconditioner}
During the local phase, MatrixPolicy and AdamW induce fixed positive definite quotient
preconditioners $P_{\MP}$ and $P_{\AW}$ satisfying
\begin{align}
    q_{t+1}^{\MP}
    &=q_t^{\MP}-\eta_{\MP}P_{\MP}\nabla F(q_t^{\MP}),
    \label{eq:mp-ideal-dynamics}\\
    q_{t+1}^{\AW}
    &=q_t^{\AW}-\eta_{\AW}P_{\AW}\nabla F(q_t^{\AW}).
    \label{eq:aw-ideal-dynamics}
\end{align}
Here $P_{\MP}$ is the quotient preconditioner after gauge rebalance and $P_{\AW}$ is the
quotient preconditioner induced by AdamW's raw-coordinate adaptive state.
\end{assumption}

\subsection{Contraction and Perturbation Bounds}
\label{app:complete-opt-proof:contraction}

For $P\succ0$, define the transformed quotient objective
\begin{equation}
    \widetilde F_P(x)=F(q_\star+P^{1/2}x).
    \label{eq:transformed-objective}
\end{equation}
A preconditioned quotient step $q^+=q-\eta P\nabla F(q)$ is equivalent to ordinary
gradient descent on $\widetilde F_P$ in the coordinate $x=P^{-1/2}(q-q_\star)$.


\begin{lemma}[Pathwise contraction certificate]
\label{lem:pathwise-contraction-certificate}
Let an ideal quotient trajectory for optimizer $o$ have nonnegative gaps
$Y_t^o=F(q_t^o)-F_\star\geq0$ and satisfy
\begin{equation}
    Y_{t+1}^o
    \leq
    \alpha_{o,t}Y_t^o,
    \qquad \alpha_{o,t}\geq0,
    \label{eq:pathwise-certificate}
\end{equation}
for $0\leq t<T$.  Then
\begin{equation}
    F(q_T^o)-F_\star
    \leq
    \left(\prod_{t=0}^{T-1}\alpha_{o,t}\right)
    \bigl(F(q_0^o)-F_\star\bigr).
    \label{eq:pathwise-certificate-product}
\end{equation}
The factors need not be below one at every step; a finite-horizon contraction follows
whenever their product is below one.
\end{lemma}

\begin{proof}
Iterate \eqref{eq:pathwise-certificate}.
\end{proof}

\begin{lemma}[Approximate quotient transfer]
\label{lem:approximate-quotient-transfer}
Let $\mathcal{J}_t$ be the actual represented loss along a trajectory and suppose
\begin{equation}
    |\mathcal{J}_t-F(q_t)|\leq\delta
\end{equation}
at $t=0$ and $t=T$.  If the quotient certificate gives
\begin{equation}
    F(q_T)-F_\star\leq A_T\bigl(F(q_0)-F_\star\bigr),
    \qquad A_T\geq0,
\end{equation}
then
\begin{equation}
    \mathcal{J}_T-F_\star
    \leq
    A_T\bigl(\mathcal{J}_0-F_\star\bigr)+(1+A_T)\delta .
    \label{eq:approximate-quotient-transfer}
\end{equation}
\end{lemma}

\begin{proof}
Using the final-time approximation error and then the quotient certificate,
\begin{equation}
    \mathcal{J}_T-F_\star
    \leq F(q_T)-F_\star+\delta
    \leq A_T(F(q_0)-F_\star)+\delta .
\end{equation}
Since $F(q_0)-F_\star\leq \mathcal{J}_0-F_\star+\delta$, the claim follows.
\end{proof}

\begin{proposition}[Expected contraction certificate]
\label{prop:expected-contraction-certificate}
Let $q_t^o$ be a stochastic quotient trajectory adapted to a filtration
$\mathcal{H}_t$, and set $Y_t^o=F(q_t^o)-F_\star\geq0$.  Suppose there are
nonnegative $\mathcal{H}_t$-measurable factors $\alpha_{o,t}$ and deterministic upper
bounds $\bar\alpha_{o,t}$ such that $\alpha_{o,t}\leq\bar\alpha_{o,t}$ almost surely and
\begin{equation}
    \mathbb{E}\left[Y_{t+1}^o\mid\mathcal{H}_t\right]
    \leq
    \alpha_{o,t}Y_t^o
    \label{eq:expected-certificate}
\end{equation}
for $0\leq t<T$.  Then
\begin{equation}
    \mathbb{E}\left[F(q_T^o)-F_\star\right]
    \leq
    \left(\prod_{t=0}^{T-1}\bar\alpha_{o,t}\right)
    \mathbb{E}\left[F(q_0^o)-F_\star\right].
    \label{eq:expected-certificate-product}
\end{equation}
The factors may be state dependent, the initial point may be random, and no individual
envelope has to be below one; finite-horizon expected contraction is controlled by the
product of the deterministic envelopes.  For two optimizers initialized from the same
distribution, the smaller product gives the smaller certified expected finite-horizon
bound.
\end{proposition}

\begin{proof}
Since $Y_t^o$ and $\alpha_{o,t}$ are $\mathcal{H}_t$-measurable and nonnegative,
\begin{equation}
    \mathbb{E}Y_{t+1}^o
    \leq
    \mathbb{E}\left[\alpha_{o,t}Y_t^o\right]
    \leq
    \bar\alpha_{o,t}\mathbb{E}Y_t^o .
\end{equation}
Iterating proves \eqref{eq:expected-certificate-product}.
\end{proof}

\begin{lemma}[One-step transformed PL decrease]
\label{lem:one-step-pl-decrease}
Let $P\succ0$, define $\widetilde F_P$ by \eqref{eq:transformed-objective}, choose
$\eta_P>0$, and set $x=P^{-1/2}(q-q_\star)$ and
$x^+=x-\eta_P\nabla\widetilde F_P(x)$.  Suppose the pathwise descent inequality
\begin{equation}
    \widetilde F_P(x^+)
    \leq
    \widetilde F_P(x)-\frac{\eta_P}{2}\norm{\nabla\widetilde F_P(x)}_2^2
    \label{eq:pathwise-descent}
\end{equation}
and the pointwise PL inequality
\begin{equation}
    \frac{1}{2}\norm{\nabla\widetilde F_P(x)}_2^2
    \geq
    \mu_P\bigl(\widetilde F_P(x)-F_\star\bigr),
    \qquad \mu_P>0,
    \qquad \eta_P\mu_P\leq1,
    \label{eq:local-pl-ineq}
\end{equation}
hold at the current iterate.  If
\begin{equation}
    q^+=q-\eta_PP\nabla F(q),
\end{equation}
then
\begin{equation}
    F(q^+)-F_\star
    \leq
    \left(1-\eta_P\mu_P\right)\bigl(F(q)-F_\star\bigr).
    \label{eq:one-step-pl-decrease}
\end{equation}
The descent condition \eqref{eq:pathwise-descent} is implied, for example, by
$L_P$-smoothness of $\widetilde F_P$ on the segment from $x$ to $x^+$ and
$0<\eta_P\leq L_P^{-1}$; global smoothness or convexity is not required.
\end{lemma}

\begin{proof}
The update in $q$ is the gradient step
$x^+=x-\eta_P\nabla\widetilde F_P(x)$.  Combining \eqref{eq:pathwise-descent} and
\eqref{eq:local-pl-ineq} gives
\begin{equation}
    \widetilde F_P(x^+)-F_\star
    \leq
    \left(1-\eta_P\mu_P\right)
    \bigl(\widetilde F_P(x)-F_\star\bigr).
\end{equation}
Since $\widetilde F_P(x)=F(q)$ and $\widetilde F_P(x^+)=F(q^+)$, this proves
\eqref{eq:one-step-pl-decrease}.
\end{proof}

\begin{proposition}[Time-varying PL quotient bound]
\label{prop:time-varying-pl-bound}
For each optimizer $o\in\{\MP,\AW\}$, suppose both ideal quotient trajectories start
from the same quotient point $q_0$ and have the form
\begin{equation}
    q_{t+1}^o=q_t^o-\eta_{o,t}P_{o,t}\nabla F(q_t^o),
    \qquad P_{o,t}\succ0,
    \qquad \eta_{o,t}>0,
    \label{eq:time-varying-quotient-update}
\end{equation}
and that Lemma~\ref{lem:one-step-pl-decrease} applies at every step with constants
$(\eta_{o,t},\mu_{o,t})$.  Define
\begin{equation}
    \Xi_o(T)=\prod_{t=0}^{T-1}\left(1-\eta_{o,t}\mu_{o,t}\right).
    \label{eq:time-varying-product}
\end{equation}
Then
\begin{equation}
    F(q_T^o)-F_\star
    \leq
    \Xi_o(T)\bigl(F(q_0)-F_\star\bigr).
    \label{eq:time-varying-pl-bound}
\end{equation}
Consequently, if $\Xi_{\MP}(T)<\Xi_{\AW}(T)$, MatrixPolicy has the smaller certified
finite-horizon PL contraction bound.
\end{proposition}

\begin{proof}
Lemma~\ref{lem:one-step-pl-decrease} gives the pathwise certificate
\eqref{eq:pathwise-certificate} with $\alpha_{o,t}=1-\eta_{o,t}\mu_{o,t}$.  Apply
Lemma~\ref{lem:pathwise-contraction-certificate}.
\end{proof}

\begin{corollary}[Frozen PL quotient bound]
\label{cor:frozen-pl-bound}
Suppose the frozen quotient dynamics in Assumption~\ref{ass:frozen-preconditioner} hold.
For $o\in\{\MP,\AW\}$, suppose Lemma~\ref{lem:one-step-pl-decrease} applies along
the transformed ideal trajectory with $P=P_o$ and fixed constants $(\eta_o,\mu_o)$.  Let
\begin{equation}
    \chi_o=1-\eta_o\mu_o.
\end{equation}
If $\eta_{\MP}\mu_{\MP}>\eta_{\AW}\mu_{\AW}$, then
\begin{equation}
    \chi_{\MP}<\chi_{\AW}
\end{equation}
and, for every initial point whose two ideal trajectories remain in their local regions,
\begin{align}
    F(q_T^{\MP})-F_\star
    &\leq \chi_{\MP}^T\bigl(F(q_0)-F_\star\bigr), \\
    F(q_T^{\AW})-F_\star
    &\leq \chi_{\AW}^T\bigl(F(q_0)-F_\star\bigr).
\end{align}
When $\eta_o=L_o^{-1}$, this recovers the condition-number form
$\chi_o=1-\kappa_o^{-1}$ with $\kappa_o=L_o/\mu_o$.
\end{corollary}

\begin{proof}
This is Proposition~\ref{prop:time-varying-pl-bound} with constant
$P_{o,t}=P_o$, $\eta_{o,t}=\eta_o$, and $\mu_{o,t}=\mu_o$.
\end{proof}

\begin{proposition}[Time-varying quadratic spectral bound]
\label{prop:time-varying-quadratic-bound}
Let
\begin{equation}
    F(q)=F_\star+\frac{1}{2}(q-q_\star)^\top H(q-q_\star),
    \qquad H\succ0,
\end{equation}
and consider the time-varying preconditioned update
\begin{equation}
    q_{t+1}=q_t-\eta_t P_t\nabla F(q_t).
\end{equation}
Define
\begin{equation}
    A_t=H^{1/2}(I-\eta_tP_tH)H^{-1/2},
    \qquad
    \alpha_t=\norm{A_t}_2^2 .
    \label{eq:time-varying-quadratic-alpha}
\end{equation}
Then
\begin{equation}
    F(q_T)-F_\star
    \leq
    \left(\prod_{t=0}^{T-1}\alpha_t\right)
    \bigl(F(q_0)-F_\star\bigr).
    \label{eq:time-varying-quadratic-bound}
\end{equation}
\end{proposition}

\begin{proof}
Let $e_t=q_t-q_\star$ and $y_t=H^{1/2}e_t$.  Since
$\nabla F(q_t)=He_t$,
\begin{equation}
    y_{t+1}=H^{1/2}(I-\eta_tP_tH)H^{-1/2}y_t=A_ty_t.
\end{equation}
Also $2(F(q_t)-F_\star)=\norm{y_t}_2^2$.  Hence
\begin{equation}
    2(F(q_{t+1})-F_\star)=\norm{A_ty_t}_2^2
    \leq \alpha_t\norm{y_t}_2^2
    =2\alpha_t(F(q_t)-F_\star).
\end{equation}
Iterating this one-step inequality gives \eqref{eq:time-varying-quadratic-bound}.
\end{proof}

\begin{lemma}[Preconditioned quadratic contraction]
\label{lem:preconditioned-contraction}
Let
\begin{equation}
    F(q)=F_\star+\frac{1}{2}(q-q_\star)^\top H(q-q_\star),
    \qquad H\succ0.
\end{equation}
For $P\succ0$, define
\begin{equation}
    K(P)=P^{1/2}HP^{1/2},
    \qquad
    \kappa(P;H)=\frac{\lammax(K(P))}{\lammin(K(P))},
    \qquad
    \rho(P;H)=\frac{\kappa(P;H)-1}{\kappa(P;H)+1}.
    \label{eq:quadratic-rho-def}
\end{equation}
With
\begin{equation}
    \eta^\star(P)=\frac{2}{\lammax(K(P))+\lammin(K(P))},
\end{equation}
the update $q_{t+1}=q_t-\eta^\star(P)P\nabla F(q_t)$ satisfies
\begin{equation}
    F(q_T)-F_\star
    \leq
    \rho(P;H)^{2T}\bigl(F(q_0)-F_\star\bigr),
    \label{eq:upper-contraction}
\end{equation}
and the bound is sharp.
\end{lemma}

\begin{proof}
Let $e_t=q_t-q_\star$ and $\widetilde e_t=P^{-1/2}e_t$.  Then
\begin{equation}
    \widetilde e_{t+1}
    =\left(I-\eta^\star(P)K(P)\right)\widetilde e_t,
    \qquad
    2(F(q_t)-F_\star)=\widetilde e_t^TK(P)\widetilde e_t.
\end{equation}
Diagonalize $K(P)=U\Lambda U^T$ and set $v_t=U^T\widetilde e_t$.  The largest multiplier
satisfies
\begin{equation}
    \max_i\left|1-\eta^\star(P)\lambda_i\right|
    =
    \frac{\lammax(K(P))-\lammin(K(P))}
    {\lammax(K(P))+\lammin(K(P))}
    =\rho(P;H).
\end{equation}
Thus
\begin{align}
    2(F(q_{t+1})-F_\star)
    &=\sum_i \lambda_i v_{t+1,i}^2 \\
    &\leq \rho(P;H)^2\sum_i \lambda_i v_{t,i}^2 \\
    &=2\rho(P;H)^2(F(q_t)-F_\star).
\end{align}
Iteration gives \eqref{eq:upper-contraction}.  Equality holds when the transformed
initial error $\widetilde e_0=P^{-1/2}(q_0-q_\star)$ is an eigenvector of $K(P)$
associated with $\lammin(K(P))$ or $\lammax(K(P))$.
\end{proof}

\begin{theorem}[Conditional worst-case optimization separation]
\label{thm:conditional-optimization-separation}
Suppose the frozen quotient dynamics in Assumption~\ref{ass:frozen-preconditioner} hold,
and suppose $F$ is the quadratic objective in Lemma~\ref{lem:preconditioned-contraction}.
Assume also that
\begin{equation}
    \kappa(P_{\MP};H)<\kappa(P_{\AW};H),
\end{equation}
and set $\eta_{\MP}=\eta^\star(P_{\MP})$ and
$\eta_{\AW}=\eta^\star(P_{\AW})$.  For every integer $T\geq1$, let $U_T$ be a
sufficiently small quotient neighborhood of $q_\star$ on which both ideal trajectories
remain in the local chart through time $T$.  Then
\begin{equation}
    \sup_{\substack{q_0\in U_T\\ q_0\neq q_\star}}
    \frac{F(q_T^{\MP})-F_\star}{F(q_0)-F_\star}
    =
    \rho(P_{\MP};H)^{2T}
    <
    \rho(P_{\AW};H)^{2T}
    =
    \sup_{\substack{q_0\in U_T\\ q_0\neq q_\star}}
    \frac{F(q_T^{\AW})-F_\star}{F(q_0)-F_\star}.
    \label{eq:main-separation}
\end{equation}
\end{theorem}

\begin{proof}
The map $\kappa\mapsto(\kappa-1)/(\kappa+1)$ is strictly increasing on the
condition-number domain $\kappa\geq1$.
The strict condition-number inequality therefore gives
$\rho(P_{\MP};H)<\rho(P_{\AW};H)$.  Lemma~\ref{lem:preconditioned-contraction} gives the
sharp worst-case factors.  Since the quadratic ratios are invariant under scaling
$q_0-q_\star$, the extremal transformed-eigenvector initializations can be chosen inside
any sufficiently small neighborhood $U_T$.
\end{proof}

\begin{assumption}[Uniform local perturbation bound]
\label{ass:uniform-perturbation}
For a fixed horizon $T$, let the ideal quotient recursions be
\begin{equation}
    q_{t+1}^o=\Psi_{o,t}(q_t^o),
    \qquad o\in\{\MP,\AW\}.
\end{equation}
The implemented quotient iterates satisfy
\begin{equation}
    q_{t+1}^{o,{\rm impl}}=\Psi_{o,t}(q_t^{o,{\rm impl}})+r_t^o,
    \qquad
    \norm{r_t^{\MP}}_2+\norm{r_t^{\AW}}_2\leq\delta
\end{equation}
for all $0\leq t<T$.  The residuals include the RMS floor, denominator floors, clipping,
moment lag, weight decay, and nonquadratic Taylor remainders.
\end{assumption}

\begin{proposition}[Stability of a strict finite-time gap]
\label{prop:complete-perturbation}
Fix $T\geq1$ and suppose the idealized dynamics satisfy
\begin{equation}
    m_T:=F(q_T^{\AW})-F(q_T^{\MP})>0 .
    \label{eq:finite-gap}
\end{equation}
Assume the implemented and ideal recursions use the same initial points, and assume the
ideal trajectories lie in a compact set $K\subset\operatorname{int}(U)$, where $U$ is a
compact neighborhood on which $F$ and all $\Psi_{o,t}$ are Lipschitz.  Then
there exists $\delta_0>0$ such that, whenever Assumption~\ref{ass:uniform-perturbation}
holds with $\delta<\delta_0$, the implemented trajectories remain in $U$ through time
$T$ and
\begin{equation}
    F(q_T^{\AW,{\rm impl}})-F(q_T^{\MP,{\rm impl}})>\frac{m_T}{2}>0 .
    \label{eq:stable-gap}
\end{equation}
\end{proposition}

\begin{proof}
Let $C_\Psi$ be a common Lipschitz constant on $U$ for the finite family of ideal
one-step maps $\{\Psi_{o,t}:o\in\{\MP,\AW\},0\leq t<T\}$.  Let
$r_{\rm stay}=\operatorname{dist}(K,U^c)>0$.
By induction, as long as the implemented trajectories remain in $U$,
\begin{equation}
    \norm{q_t^{\MP,{\rm impl}}-q_t^{\MP}}
    +
    \norm{q_t^{\AW,{\rm impl}}-q_t^{\AW}}
    \leq C_T\delta
\end{equation}
where one may take
\begin{equation}
    C_T=\sum_{j=0}^{T-1}C_\Psi^j,
\end{equation}
with the usual value $C_T=T$ when $C_\Psi=1$.  Taking
$\delta\leq r_{\rm stay}/(2C_T)$ keeps the implemented trajectories in $U$ and
closes the induction.  If $C_F$ is a Lipschitz constant for $F$ on $U$, then
\begin{equation}
    \left|F(q_T^{\MP,{\rm impl}})-F(q_T^{\MP})\right|
    +
    \left|F(q_T^{\AW,{\rm impl}})-F(q_T^{\AW})\right|
    \leq C_F C_T\delta .
\end{equation}
Taking
\begin{equation}
    \delta_0=\min\left\{\frac{m_T}{2C_FC_T},\frac{r_{\rm stay}}{2C_T}\right\}
\end{equation}
gives the claim.
\end{proof}
